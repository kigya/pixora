cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(PixoraRecreate LANGUAGES C CXX)
endif()

if(NOT TARGET stable-diffusion)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/stable-diffusion.cpp/CMakeLists.txt")
    message(FATAL_ERROR "Pinned stable-diffusion.cpp submodule is missing; initialize submodules recursively")
  endif()
  # Configure CPU-only options in a function scope so these choices apply only to
  # the newly added diffusion dependency, not sibling app targets.
  function(pixora_add_stable_diffusion_vendor)
    cmake_policy(SET CMP0077 NEW)
    set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
    set(SD_BUILD_EXAMPLES OFF)
    set(SD_BUILD_TESTS OFF)
    set(SD_BUILD_SHARED_LIBS OFF)
    set(SD_METAL OFF)
    set(SD_VULKAN OFF)
    set(GGML_NATIVE OFF)
    set(GGML_OPENMP OFF)
    set(GGML_METAL OFF)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/vendor/stable-diffusion.cpp" stable-diffusion-vendor)
  endfunction()
  pixora_add_stable_diffusion_vendor()
endif()

add_library(pixora-recreate STATIC recreate.cpp)
target_include_directories(pixora-recreate PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(pixora-recreate PUBLIC stable-diffusion)
target_compile_features(pixora-recreate PUBLIC cxx_std_17)

if(PIXORA_RECREATE_BUILD_CLI)
  add_executable(recreate-cli cli.cpp)
  target_link_libraries(recreate-cli PRIVATE pixora-recreate)
endif()

if(PIXORA_RECREATE_BUILD_TESTS)
  enable_testing()
  add_executable(recreate-api-test tests/recreate_api_test.cpp)
  target_link_libraries(recreate-api-test PRIVATE pixora-recreate)
  add_test(NAME recreate-api-test COMMAND recreate-api-test)
endif()
